153 template <typename Cycle, typename Mapping, typename Ratio, typename Direction, typename Round,
154 typename OnCycle>
155 auto parametric_search(Mapping& dist, Ratio ratio, Direction direction, Round& round,
156 OnCycle on_cycle, bool first_only = false) -> std::pair<Ratio, Cycle> {
157 auto ratio_best = ratio;
158 auto cycle_best = Cycle{};
159 auto cycle = Cycle{};
160
161 while (true) {
162 auto visit = [&](auto&& candidate) -> bool {
163 auto ratio_i = on_cycle(candidate);
164 if (!direction.better(ratio_i, ratio_best)) return true;
165 ratio_best = ratio_i;
166 cycle_best = std::forward<decltype(candidate)>(candidate);
167 return !first_only;
168 };
169 round(dist, ratio, visit);
170
171 if (direction.converged(ratio_best, ratio)) break;
172
173 cycle = std::move(cycle_best);
174 ratio = ratio_best;
175 }
176
177 return {ratio, std::move(cycle)};
178 }
