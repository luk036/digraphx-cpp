101 auto _find_cycle(const absl::flat_hash_map<Node, std::pair<Node, Edge>>& point_to)
102 -> py::Generator<Node> {
103 auto visited = absl::flat_hash_map<Node, Node>{};
104 if constexpr (requires { this->_digraph.size(); }) visited.reserve(this->_digraph.size());
105 for (const auto& entry : this->_digraph) {
106 const auto& vtx = _get_key(entry);
107 if (visited.contains(vtx)) continue;
108 auto utx = vtx;
109 visited[utx] = vtx;
110 while (point_to.contains(utx)) {
111 utx = point_to.at(utx).first;
112 auto it = visited.find(utx);
113 if (it != visited.end()) {
114 if (it->second == vtx) co_yield utx;
115 break;
116 }
117 visited[utx] = vtx;
118 }
119 }
120 co_return;
121 }

123 template <typename Mapping, typename GetWeight, typename UpdateOk>
124 auto _relax_pred(Mapping& dist, GetWeight&& get_weight, UpdateOk&& update_ok) -> bool {
125 auto changed = false;
126 for (const auto& entry : this->_digraph) {
127 const auto& utx = _get_key(entry);
128 const auto& nbrs = _get_val(entry, this->_digraph);
129 for (const auto& nbr_entry : nbrs) {
130 const auto& vtx = _get_key(nbr_entry);
131 const auto& edge = _get_val(nbr_entry, nbrs);
132 auto distance = dist[utx] + std::forward<GetWeight>(get_weight)(edge);
133 if (dist[vtx] > distance
134 && std::forward<UpdateOk>(update_ok)(dist[vtx], distance)) {
135 dist[vtx] = distance;
136 this->_pred.insert_or_assign(vtx, std::pair(utx, edge));
137 changed = true;
138 }
139 }
140 }
141 return changed;

165 auto _cycle_list(const Node& handle,
166 const absl::flat_hash_map<Node, std::pair<Node, Edge>>& point_to) const
167 -> Cycle {
168 auto vtx = handle;
169 auto cycle = Cycle{};
170 cycle.reserve(point_to.size());
171 while (true) {
172 const auto& [utx, edge] = point_to.at(vtx);
173 cycle.emplace_back(edge);
174 vtx = utx;
175 if (vtx == handle) break;
176 }
177 return cycle;
