31 // Get the key from an iteration element:
32 // - For pair-like (unordered_map, list<pair>): .first
33 // - For direct (SimpleDiGraphS nodes): the element itself
34 template <typename T> decltype(auto) _get_key(const T& entry) {
35 if constexpr (requires { entry.first; }) {
36 return entry.first;
37 } else {
38 return entry;
39 }

42 // Get the value from an iteration element:
43 // - For pair-like: .second
44 // - For direct: .at(key) on the container
45 template <typename T, typename Container>
46 decltype(auto) _get_val(const T& entry, const Container& c) {
47 if constexpr (requires { entry.second; }) {
48 return entry.second;

83 template <typename DiGraph, typename PointTo>
84 auto find_cycle(const DiGraph& digraph, const PointTo& point_to)
85 -> py::Generator<typename graph_traits<DiGraph>::Node> {
86 using Node = typename graph_traits<DiGraph>::Node;
87 auto visited = absl::flat_hash_map<Node, Node>{};
88 if constexpr (requires { digraph.size(); }) visited.reserve(digraph.size());
89 for (const auto& entry : digraph) {
90 const auto& vtx = _get_key(entry);
91 // Single probe: mark the DFS root and skip if already visited.
92 if (!visited.try_emplace(vtx, vtx).second) continue;
93 auto utx = vtx;
94 while (true) {
95 auto pit = point_to.find(utx);
96 if (pit == point_to.end()) break;
97 utx = pit->second.first;
98 auto it = visited.find(utx);
99 if (it != visited.end()) {
100 if (it->second == vtx) co_yield utx;
101 break;
102 }
103 visited.emplace(utx, vtx);
104 }
105 }
106 co_return;
107 }

117 template <typename PointTo>
118 auto cycle_list(const typename PointTo::key_type& handle, const PointTo& point_to)
119 -> std::vector<typename PointTo::mapped_type::second_type> {
120 using Edge = typename PointTo::mapped_type::second_type;
121 auto vtx = handle;
122 auto cycle = std::vector<Edge>{};
123 cycle.reserve(point_to.size());
124 while (true) {
125 const auto& [utx, edge] = point_to.at(vtx);
126 cycle.emplace_back(edge);
127 vtx = utx;
128 if (vtx == handle) break;
129 }
130 return cycle;
131 }

198 auto relax_pred(const DiGraph& digraph, Mapping& dist, GetWeight&& get_weight, PointTo& pred,
199 UpdateOk&& update_ok) -> bool {
200 auto changed = false;
201 for (const auto& entry : digraph) {
202 const auto& utx = _get_key(entry);
203 const auto& nbrs = _get_val(entry, digraph);
204 for (const auto& nbr_entry : nbrs) {
205 const auto& vtx = _get_key(nbr_entry);
206 const auto& edge = _get_val(nbr_entry, nbrs);
207 auto distance = dist[utx] + std::forward<GetWeight>(get_weight)(edge);
208 if (dist[vtx] > distance
209 && std::forward<UpdateOk>(update_ok)(dist[vtx], distance)) {
210 dist[vtx] = distance;
211 pred.insert_or_assign(vtx, std::pair(utx, edge));
212 changed = true;
213 }
214 }
215 }
216 return changed;

282 template <typename GetWeight> auto operator()(Mapping& dist, GetWeight& get_weight)
283 -> bool {
284 const auto build = (_pass == 1);
285 if (build) _weights.clear();
286 auto wit = _weights.begin();
287 auto changed = false;
288 for (const auto& entry : _digraph) {
289 const auto& utx = _get_key(entry);
290 const auto& nbrs = _get_val(entry, _digraph);
291 for (const auto& nbr_entry : nbrs) {
292 const auto& vtx = _get_key(nbr_entry);
293 const auto& edge = _get_val(nbr_entry, nbrs);
294 const auto weight
295 = (_pass >= 2) ? *wit++ : static_cast<Weight>(get_weight(edge));
296 if (build) _weights.push_back(weight);
297 if constexpr (IsPred) {
298 auto distance = dist[utx] + weight;
299 if (dist[vtx] > distance && _update_ok(dist[vtx], distance)) {
300 dist[vtx] = distance;
301 _point_to.insert_or_assign(vtx, std::pair(utx, edge));
302 changed = true;
303 }
304 } else {
305 auto distance = dist[vtx] - weight;
306 if (dist[utx] < distance && _update_ok(dist[utx], distance)) {
307 dist[utx] = distance;
308 _point_to.insert_or_assign(utx, std::pair(vtx, edge));
309 changed = true;
310 }
311 }
312 }
313 }
314 ++_pass;
315 return changed;
316 }

362 template <typename DiGraph, typename Mapping, typename GetWeight, typename PointTo,
363 typename Relax>
364 auto howard_search(const DiGraph& digraph, Mapping& dist, GetWeight get_weight,
365 PointTo& point_to, Relax relax)
366 -> py::Generator<typename graph_traits<DiGraph>::Cycle> {
367 point_to.clear();
368 if constexpr (requires { digraph.size(); }) point_to.reserve(digraph.size());
369 auto found = false;
370 while (!found && relax(dist, get_weight)) {
371 for (const auto& vtx : find_cycle(digraph, point_to)) {
372 assert(relax.is_negative(vtx, dist, get_weight));
373 co_yield cycle_list(vtx, point_to);
374 found = true;
375 }
376 }
377 co_return;
378 }

406 auto zero_cancel(const std::vector<Edge>& cycle) const -> Ratio {
407 auto total_cost = Ratio{0};
408 auto total_time = Ratio{0};
409 for (const auto& edge : cycle) {
410 const auto [cost, time] = _cost_time(edge);
411 total_cost += static_cast<Ratio>(cost);
412 total_time += static_cast<Ratio>(time);
413 }
414 if (total_time == Ratio{0}) return std::numeric_limits<Ratio>::infinity();
415 return total_cost / total_time;
416 }
417
418 private:
419 CostTime _cost_time;
420 };
