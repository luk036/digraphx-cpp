82 template <typename DiGraph, typename PointTo>
83 auto find_cycle(const DiGraph& digraph, const PointTo& point_to)
84 -> py::Generator<typename graph_traits<DiGraph>::Node> {
85 using Node = typename graph_traits<DiGraph>::Node;
86 auto visited = absl::flat_hash_map<Node, Node>{};
87 if constexpr (requires { digraph.size(); }) visited.reserve(digraph.size());
88 for (const auto& entry : digraph) {
89 const auto& vtx = _get_key(entry);
90 if (visited.contains(vtx)) continue;
91 auto utx = vtx;
92 visited[utx] = vtx;
93 while (point_to.contains(utx)) {
94 utx = point_to.at(utx).first;
95 auto it = visited.find(utx);
96 if (it != visited.end()) {
97 if (it->second == vtx) co_yield utx;
98 break;
99 }
100 visited[utx] = vtx;
101 }
102 }
103 co_return;
104 }

114 template <typename PointTo>
115 auto cycle_list(const typename PointTo::key_type& handle, const PointTo& point_to)
116 -> std::vector<typename PointTo::mapped_type::second_type> {
117 using Edge = typename PointTo::mapped_type::second_type;
118 auto vtx = handle;
119 auto cycle = std::vector<Edge>{};
120 cycle.reserve(point_to.size());
121 while (true) {
122 const auto& [utx, edge] = point_to.at(vtx);
123 cycle.emplace_back(edge);
124 vtx = utx;
125 if (vtx == handle) break;
126 }
127 return cycle;
128 }

174 auto relax_pred(const DiGraph& digraph, Mapping& dist, GetWeight&& get_weight, PointTo& pred,
175 UpdateOk&& update_ok) -> bool {
176 auto changed = false;
177 for (const auto& entry : digraph) {
178 const auto& utx = _get_key(entry);
179 const auto& nbrs = _get_val(entry, digraph);
180 for (const auto& nbr_entry : nbrs) {

184 if (dist[vtx] > distance

186 dist[vtx] = distance;
187 pred.insert_or_assign(vtx, std::pair(utx, edge));
188 changed = true;
189 }
190 }
191 }
192 return changed;

254 template <typename DiGraph, typename Mapping, typename GetWeight, typename PointTo,
255 typename Relax, typename Check>
256 auto howard_search(const DiGraph& digraph, Mapping& dist, GetWeight get_weight,
257 PointTo& point_to, Relax relax, Check check)
258 -> py::Generator<typename graph_traits<DiGraph>::Cycle> {
259 using Cycle = typename graph_traits<DiGraph>::Cycle;
260 point_to.clear();
261 if constexpr (requires { digraph.size(); }) point_to.reserve(digraph.size());
262 auto found = false;
263 while (!found && relax(dist, get_weight)) {
264 for (const auto& vtx : find_cycle(digraph, point_to)) {
265 check(vtx, dist, get_weight);

268 }
