228 template <typename Mapping> auto run(Mapping& dist, Ratio ratio, const UpdateOk& update_ok,
229 bool pick_one_only = false) -> std::pair<Ratio, Cycle> {
230 // Helper function to calculate edge weights based on current ratio
231 auto get_weight = [this, &ratio](const Edge& edge) -> Domain {
232 return static_cast<Domain>(this->_omega.distance(ratio, edge));
233 };
234
235 // Initialize tracking variables for maximum ratio and corresponding cycle
236 auto r_max = ratio;
237 auto c_max = Cycle{};
238 auto cycle = Cycle{};
239 auto reverse = true; // Flag to alternate search direction
240
241 // Initialize the negative cycle finder with our graph
242 auto ncf = NegCycleFinderQ<DiGraph, Domain>{this->_digraph};
243
244 // Main optimization loop
245 while (true) {
246 // Search for cycles in either forward or reverse direction
247 if (reverse) {
248 auto cycles = ncf.howard_succ(dist, get_weight, update_ok);
249 for (auto&& c_i : cycles) {
250 auto r_i = this->_omega.zero_cancel(c_i);
251 if (r_max < r_i) {
252 r_max = r_i;
253 c_max = std::move(c_i);
254 if (pick_one_only) { // Early exit if we only need one improvement
255 break;
256 }
257 }
258 }
259 } else {
260 auto cycles = ncf.howard_pred(dist, get_weight, update_ok);
261 for (auto&& c_i : cycles) {
262 auto r_i = this->_omega.zero_cancel(c_i);
263 if (r_max < r_i) {
264 r_max = r_i;
265 c_max = std::move(c_i);
266 if (pick_one_only) { // Early exit if we only need one improvement
267 break;
268 }
269 }
270 }
271 }
272
273 // Termination condition: no better ratio found
274 if (r_max <= ratio) {
275 break;
276 }
277
278 // Update state for next iteration
279 cycle = std::move(c_max);
280 ratio = r_max;
281 reverse = !reverse; // Alternate search direction
282 }
283
284 return std::make_pair(ratio, std::move(cycle));
285 }
