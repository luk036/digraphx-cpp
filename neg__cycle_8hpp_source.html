45 // Get the key from an iteration element:

140 template <typename Mapping, typename Callable> auto _relax(Mapping& dist, Callable&& get_weight)
141 -> bool {
142 auto changed = false;
143 for (const auto& entry : this->_digraph) {
144 const auto& utx = _get_key(entry);
145 const auto& nbrs = _get_val(entry, this->_digraph);
146 for (const auto& nbr_entry : nbrs) {
147 const auto& vtx = _get_key(nbr_entry);
148 const auto& edge = _get_val(nbr_entry, nbrs);
149 auto distance = dist[utx] + std::forward<Callable>(get_weight)(edge);
150 if (dist[vtx] > distance) {
151 dist[vtx] = distance;
152 this->_pred.insert_or_assign(vtx, std::pair(utx, edge));
153 changed = true;
154 }

232 auto _find_cycle() -> py::Generator<Node> {
233 auto visited = absl::flat_hash_map<Node, Node>{};
234 if constexpr (requires { this->_digraph.size(); }) visited.reserve(this->_digraph.size());
235 for (const auto& entry : this->_digraph) {
236 const auto& vtx = _get_key(entry);
237 if (visited.contains(vtx)) continue;
238 auto utx = vtx;
239 visited[utx] = vtx;
240 while (this->_pred.contains(utx)) {
241 utx = this->_pred[utx].first;
242 auto it = visited.find(utx);
243 if (it != visited.end()) {
244 if (it->second == vtx) co_yield utx;
245 break;
246 }
247 visited[utx] = vtx;
248 }
249 }
250 co_return;
